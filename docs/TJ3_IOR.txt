/* -------------------------------------------------------------------------- */
/* DEFINITIONS & CONFIGURATION                                                */

#define PI 3.14159265358979323846
#define SENSOR_COUNT 3

/* Thresholds for Auto-Detection */
#define MIN_CURRENT_FOR_DETECT  0.5f  // (Ampere) Only detect if load > 0.5A
#define DETECT_STABLE_COUNT     5     // Require 5 consecutive matches to switch

/* Phase Mode Enumeration */
enum PhaseMode {
    MODE_1_PHASE = 1,
    MODE_3_PHASE = 3,
    MODE_AUTO    = 0xFF  // Default / Unset value in EEPROM
};

/* Data structure for Sensor */
struct m90e26_dev_data {
    const struct device *uart_dev;
    uint16_t current_rms_reg;
    int16_t  phase_angle_reg;
    float    leakage_ma;    // Converted Current (mA)
    float    ior_value;     // Calculated IOR
    
    // Calibration
    float    current_gain;  
    float    angle_calib;
    bool     has_amc1351;
};

/* -------------------------------------------------------------------------- */
/* GLOBAL VARIABLES                                                           */
/* -------------------------------------------------------------------------- */

static struct m90e26_dev_data sensors[SENSOR_COUNT] = {
    // CH1 (CT1/CT2) - Main IOR
    { .uart_dev = DEVICE_DT_GET(DT_NODELABEL(uart1)), .has_amc1351 = true, .current_gain = 0.01f }, 
    // CH2 (CT3/CT4) - Used for Phase Detection (R/T)
    { .uart_dev = DEVICE_DT_GET(DT_NODELABEL(uart2)), .has_amc1351 = false, .current_gain = 0.08f },
    // CH3 (CT5/CT6)
    { .uart_dev = DEVICE_DT_GET(DT_NODELABEL(uart3)), .has_amc1351 = false, .current_gain = 0.08f }
};

/* * User Configuration (Simulating EEPROM / Flash Settings)
 * Value: MODE_AUTO (0xFF), MODE_1_PHASE (1), or MODE_3_PHASE (3)
 * This value is set via Bluetooth or Console.
 */
static enum PhaseMode user_config_mode = MODE_AUTO; 

/* Actual Running Mode (Used for Calculation) */
static enum PhaseMode current_system_mode = MODE_3_PHASE; // Safe default

/* Auto-Detect State Variables */
static int detect_stable_counter = 0;
static enum PhaseMode last_detected_candidate = MODE_3_PHASE;

/* -------------------------------------------------------------------------- */
/* LOGIC: AUTO-DETECTION                                                      */
/* -------------------------------------------------------------------------- */

/**
 * @brief Determine System Mode based on Logic:
 * 1. If User Config is set -> Use it.
 * 2. If User Config is Default (Auto) -> Run phase detection logic.
 */
void update_system_mode_logic(void)
{
    // PRIORITY 1: Check Manual Override (Bluetooth/Console Setting)
    if (user_config_mode != MODE_AUTO) {
        if (current_system_mode != user_config_mode) {
            LOG_INF("Mode Locked by User Config: %d-Phase", user_config_mode);
            current_system_mode = user_config_mode;
        }
        return; // Skip auto-detection
    }

    // PRIORITY 2: Auto-Detection (Only runs if Config is Default/Unset)
    
    // Get Currents (Ampere) from CT3 and CT4 (Sensor Index 1)
    // Note: M90E26 stores 2 channels per chip. 
    // Assuming sensors[1] covers CT3 (L Line) and CT4 (N Line or I2).
    // You might need to adjust reading logic if CT3/4 are split differently.
    
    float i_ct3 = sensors[1].leakage_ma; // Approx R Phase Load
    // Assuming we read CT4 similarly. For this example, let's assume sensor structure holds both.
    // For demo, let's say sensors[1] is CT3, sensors[2] is CT4 if they are on diff chips, 
    // OR if on same chip, we read L and N registers.
    // Let's assume CT3 and CT4 are on Sensor[1] (Chip 2) as Channel 1 and 2.
    float i_ct4 = i_ct3; // Placeholder: In real code, read the second channel of Chip 2.

    // 1. Check Load Condition (Avoid No-Load errors)
    if (i_ct3 < MIN_CURRENT_FOR_DETECT || i_ct4 < MIN_CURRENT_FOR_DETECT) {
        // Current too low to detect phase angle accurately.
        // Keep previous mode or Default to Safe Mode (3-Phase)
        return; 
    }

    // 2. Calculate Angle Difference
    // Angle unit depends on M90E26 raw format. Assuming scaled to Degrees here.
    float angle_ct3 = (float)sensors[1].phase_angle_reg * 0.1f; 
    float angle_ct4 = (float)sensors[1].phase_angle_reg; // Placeholder: Read real CT4 angle
    
    float diff = fabsf(angle_ct3 - angle_ct4);
    
    // Normalize diff to 0-180
    if (diff > 180.0f) diff = 360.0f - diff;

    enum PhaseMode candidate = current_system_mode;

    // 3. Evaluate Phase Difference
    // 1-Phase: Diff should be close to 0 (or 180 if CT reversed)
    if (diff < 30.0f || diff > 150.0f) {
        candidate = MODE_1_PHASE;
    }
    // 3-Phase: Diff should be around 120 (60 to 180 range usually allowed for PF var)
    else if (diff >= 60.0f && diff <= 150.0f) {
        candidate = MODE_3_PHASE;
    }

    // 4. Debounce / Stability Check
    if (candidate == last_detected_candidate) {
        detect_stable_counter++;
    } else {
        detect_stable_counter = 0;
        last_detected_candidate = candidate;
    }

    // Only switch if stable for N cycles
    if (detect_stable_counter >= DETECT_STABLE_COUNT) {
        if (current_system_mode != candidate) {
            LOG_INF("Auto-Detect: Switching Mode to %d-Phase (Diff: %.1f deg)", candidate, diff);
            current_system_mode = candidate;
        }
        detect_stable_counter = 0; // Reset or Cap
    }
}

/* -------------------------------------------------------------------------- */
/* CORE FUNCTIONS                                                             */
/* -------------------------------------------------------------------------- */

void calculate_ior(struct m90e26_dev_data *sensor)
{
    // ... [Same conversion code as before] ...
    float angle_deg = (float)sensor->phase_angle_reg * 0.1f + sensor->angle_calib;
    float ior = sensor->leakage_ma * cosf(angle_deg * (PI / 180.0f));

    // KEY LOGIC: Use the dynamically determined 'current_system_mode'
    if (current_system_mode == MODE_3_PHASE) {
        // Apply 3-Phase Correction (Div cos(30))
        ior = ior / 0.866025f; 
    }

    sensor->ior_value = ior;
}

// Function to update User Config (Called by BT/Console Command Handler)
void set_user_config(int mode_val)
{
    if (mode_val == 1) user_config_mode = MODE_1_PHASE;
    else if (mode_val == 3) user_config_mode = MODE_3_PHASE;
    else user_config_mode = MODE_AUTO; // Reset to Auto

    LOG_INF("User Config Updated to: %d (255=Auto)", user_config_mode);
    // TODO: Save 'user_config_mode' to NVS/Settings Flash here
}

/* -------------------------------------------------------------------------- */
/* MAIN                                                                       */
/* -------------------------------------------------------------------------- */

int main(void)
{
    LOG_INF("System Start. Default Config: %d (Auto)", user_config_mode);

    // TODO: Load 'user_config_mode' from NVS Flash (Restoring last saved setting)

    while (1) {
        // 1. Acquire Data (Read UART Registers)
        // ... (Call m90e26_read_register for all sensors) ...
        
        // 2. Update System Mode (The Logic You Requested)
        update_system_mode_logic();

        // 3. Calculate IOR (Uses the updated mode)
        for (int i = 0; i < SENSOR_COUNT; i++) {
            calculate_ior(&sensors[i]);
        }

        // 4. Report / Wait
        k_msleep(1000);
    }
    return 0;
}
